from fpioa_manager import fm
from machine import Timer, PWM

tim1 = Timer(Timer.TIMER0, Timer.CHANNEL0, mode=Timer.MODE_PWM)
PWM1 = PWM(tim1, freq=500000, duty=50, pin=21)
tim2 = Timer(Timer.TIMER0, Timer.CHANNEL1, mode=Timer.MODE_PWM)
PWM2 = PWM(tim2, freq=500000, duty=50, pin=22)
tim3 = Timer(Timer.TIMER0, Timer.CHANNEL2, mode=Timer.MODE_PWM)
PWM3 = PWM(tim3, freq=500000, duty=50, pin=23)
tim4 = Timer(Timer.TIMER0, Timer.CHANNEL3, mode=Timer.MODE_PWM)
PWM4 = PWM(tim4, freq=500000, duty=50, pin=24)
tim5 = Timer(Timer.TIMER1, Timer.CHANNEL0, mode=Timer.MODE_PWM)
PWM5 = PWM(tim5, freq=500000, duty=50, pin=15)
tim6 = Timer(Timer.TIMER1, Timer.CHANNEL1, mode=Timer.MODE_PWM)
PWM6 = PWM(tim6, freq=500000, duty=50, pin=32)
tim7 = Timer(Timer.TIMER1, Timer.CHANNEL2, mode=Timer.MODE_PWM)
PWM7 = PWM(tim7, freq=500000, duty=50, pin=14)
tim8 = Timer(Timer.TIMER1, Timer.CHANNEL3, mode=Timer.MODE_PWM)
PWM8 = PWM(tim8, freq=500000, duty=50, pin=13)
def car_forward(Motor_A,Motor_B,Motor_C,Motor_D):
    PWM1.duty(0)
    PWM2.duty(Motor_A)
    PWM3.duty(0)
    PWM4.duty(Motor_B)
    PWM5.duty(0)
    PWM6.duty(Motor_C)
    PWM7.duty(0)
    PWM8.duty(Motor_D)
def car_backward(Motor_A,Motor_B,Motor_C,Motor_D):
    PWM1.duty(Motor_A)
    PWM2.duty(0)
    PWM3.duty(Motor_B)
    PWM4.duty(0)
    PWM5.duty(Motor_C)
    PWM6.duty(0)
    PWM7.duty(Motor_D)
    PWM8.duty(0)
def car_turn_left(Motor_A,Motor_B,Motor_C,Motor_D):
    PWM1.duty(Motor_A)
    PWM2.duty(0)
    PWM3.duty(0)
    PWM4.duty(Motor_B)
    PWM5.duty(Motor_C)
    PWM6.duty(0)
    PWM7.duty(0)
    PWM8.duty(Motor_D)
def car_turn_right(Motor_A,Motor_B,Motor_C,Motor_D):
    PWM1.duty(0)
    PWM2.duty(Motor_A)
    PWM3.duty(Motor_B)
    PWM4.duty(0)
    PWM5.duty(0)
    PWM6.duty(Motor_C)
    PWM7.duty(Motor_D)
    PWM8.duty(0)
def car_left(Motor_A,Motor_B,Motor_C,Motor_D):
    PWM1.duty(Motor_A)
    PWM2.duty(0)
    PWM3.duty(0)
    PWM4.duty(Motor_B)
    PWM5.duty(0)
    PWM6.duty(Motor_C)
    PWM7.duty(Motor_D)
    PWM8.duty(0)
def car_right(Motor_A,Motor_B,Motor_C,Motor_D):
    PWM1.duty(0)
    PWM2.duty(Motor_A)
    PWM3.duty(Motor_B)
    PWM4.duty(0)
    PWM5.duty(Motor_C)
    PWM6.duty(0)
    PWM7.duty(0)
    PWM8.duty(Motor_D)

Servo1_tim = Timer(Timer.TIMER2, Timer.CHANNEL0, mode=Timer.MODE_PWM)
S1 = PWM(Servo1_tim, freq=50, duty=0, pin=12)
Servo2_tim = Timer(Timer.TIMER2, Timer.CHANNEL1, mode=Timer.MODE_PWM)
S2 = PWM(Servo2_tim, freq=50, duty=0, pin=11)
Servo3_tim = Timer(Timer.TIMER2, Timer.CHANNEL2, mode=Timer.MODE_PWM)
S3 = PWM(Servo3_tim, freq=50, duty=0, pin=10)
Servo4_tim = Timer(Timer.TIMER2, Timer.CHANNEL3, mode=Timer.MODE_PWM)
S4 = PWM(Servo4_tim, freq=50, duty=0, pin=3)
def Servo_1(angle):
    S1.duty((angle+90)/180*10+2.5)
def Servo_2(angle):
    S2.duty((angle+90)/180*10+2.5)
def Servo_3(angle):
    S3.duty((angle+90)/180*10+2.5)
def Servo_4(angle):
    S4.duty((angle+90)/180*10+2.5)

#启动后停车，舵机归位
