#-------------------------------识别不同颜色的色块------------------------------------#
def find_blob(color_threshold,roi_area):
    img = sensor.snapshot()
    blobs = img.find_blobs([color_threshold],roi=roi_area)  #roi为感兴趣区域，可以控制图像处理的区域
    global b  #设为全局变量
    if blobs:
        for b in blobs:
            if b[4] > 100:    #b[4]为像素点总数，这个判断用于去除一些不需要的干扰色块
                tmp = img.draw_rectangle(b[0:4],color = (255, 255, 0))   #画框
                tmp = img.draw_cross(b[5],b[6],color = (255, 255, 0))   #画像素中心点
                print(b)
                print(b[5])  #b[5]是色块中心点像素的x轴位置，可以用于巡线
                #print(b[6])   #b[6]是色块中心点y轴位置，判断远近]
    lcd.display(img)

##################此程序设置算法，根据跑道情况，进行修改。################
#为了精确寻迹，可以将摄像头阻尼铰链向下(看得近)
#为了提高巡线速度，可以将摄像头阻尼铰向上转(看得远)
#1、修改black_threshold，用于适应环境灯光亮度。具体方法参考【视频教程2-颜色阈值确定方法】
#2、k为调整转向灵敏度参数，转向越急，k调大，转向越灵敏。
#3、track_line_area，巡黑线的视觉区域，尽可能显示在显示器中间。范围太大容易识别到远处的其他黑线，范围太小不能及时捕捉到转弯线。
#【x轴左侧起点，y轴上面起点，x轴宽度区域，y轴高度区域】以x轴160，y轴120为中心。
#4、p为整个寻迹速度，为了保证pwm>50，电机启动，p必须大于0.75，为了保证pwm<=100，p必须小于等于1
black_threshold = (0, 39, -128, 127, -128, 127)
k=80
track_line_area = [60,80,200,80]
p=1


while True:
    clock.tick()
    find_blob(black_threshold,track_line_area)   #找到黑色色块，find_blob()函数会获得一个返回值b，b是一个数组，包含了色块的起始坐标、宽度、像素数量、像素中心坐标
    d_speed = (160 - b[5])    ##b[5]是像素的中心位置，160是整个画面320*240像素的一半位置，建立一个变量d_speed用于记录与中心黑线的距离差
    k_speed=d_speed*25/(160-k) #为了保证pwm不能超过100或小于0，故b_speed和k_speed需要建立关系。
    if k_speed>25: k_speed=25
    elif k_speed<-25 :k_speed=-25
    car_forward(p*(75-k_speed),p*(75+k_speed),p*(75-k_speed),p*(75+k_speed))
    #print("fps={}".format(clock.fps()))
